def ask_gemini(user_message):
    """Lightning-fast Gemini call optimized for speed."""
    try:
        model = genai.GenerativeModel("gemini-1.5-flash")
        prompt = f"{build_system_prompt()}\n\nCustomer Message: {user_message}"
        
        response = model.generate_content(
            prompt,
            generation_config={"max_output_tokens": 150, "temperature": 0.3}
        )
        
        if response and hasattr(response, 'text') and response.text:
            return response.text.strip()
        else:
            return "Hello! How can I assist you with our inventory today?"
            
    except Exception as e:
        logging.error(f"Error querying Gemini API: {e}")
        return "Hello! Please let me know how I can help you."

def send_whatsapp_msg(to_number, message):
    """Send message via UltraMsg with short timeout to prevent blocking."""
    try:
        url = f"https://api.ultramsg.com/{ULTRAMSG_INSTANCE_ID}/messages/chat"
        payload = {
            "token": ULTRAMSG_TOKEN,
            "to": to_number,
            "body": message
        }
        headers = {'content-type': 'application/x-www-form-urlencoded'}
        requests.post(url, data=payload, headers=headers, timeout=5)
    except Exception as e:
        logging.error(f"Failed to send WhatsApp message: {e}")

@app.route('/webhook', methods=['POST'])
def webhook():
    global CURRENT_MODE
    try:
        data = request.get_json(silent=True)
        if not data or 'data' not in data:
            return jsonify({"status": "ignored"}), 200

        msg_data = data['data']
        if 'body' not in msg_data or not msg_data['body']:
            return jsonify({"status": "ignored"}), 200

        incoming_msg = str(msg_data['body']).strip()
        from_number = msg_data.get('from', '')

        if not incoming_msg or not from_number:
            return jsonify({"status": "ignored"}), 200

        # Instant Mode Switch Commands
        cmd = incoming_msg.lower()
        if cmd == "!mode mobile":
            CURRENT_MODE = "MOBILE (Electronics & Retail Store)"
            send_whatsapp_msg(from_number, "✅ Mode switched to: Retail & Mobile Shop")
            return jsonify({"status": "success"}), 200
        elif cmd == "!mode estate":
            CURRENT_MODE = "ESTATE (Real Estate Agency)"
            send_whatsapp_msg(from_number, "✅ Mode switched to: Real Estate Agency")
            return jsonify({"status": "success"}), 200
        elif cmd == "!mode food":
            CURRENT_MODE = "FOOD (Restaurant & Delivery)"
            send_whatsapp_msg(from_number, "✅ Mode switched to: Food & Restaurant Service")
            return jsonify({"status": "success"}), 200

        # Process AI Reply
        ai_reply = ask_gemini(incoming_msg)
        send_whatsapp_msg(from_number, ai_reply)

    except Exception as e:
        logging.error(f"Webhook Error: {e}")

    return jsonify({"status": "success"}), 200
